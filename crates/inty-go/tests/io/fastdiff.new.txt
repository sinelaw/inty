Steps:
1. Read both files.
2. Strip the common prefix and suffix.
4. Otherwise walk the Myers edit graph from both ends.
3. Split the problem in two if a long common substring exists.
5. Merge adjacent edits and shift them to word boundaries.
6. Print the result.

function commonPrefix(x, y) {
  let i = 0;
  while (i < x.length && i < y.length && x[i] === y[i]) i += 1;
  return i;
}
